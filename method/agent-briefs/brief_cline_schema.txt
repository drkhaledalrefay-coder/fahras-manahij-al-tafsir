RECORD SHAPE (JSON Schema draft 2020-12; all fields below required unless marked optional):
{
 "annotation_id": string, e.g. "ibn_kathir-2_255_tafsir-m13",
 "quran": {"surah": int 1..114, "ayah": int >=1, "ayah_end": optional int},
 "tafsir": {"id": string slug, "name": string},
 "source": {"source_id": string, "sha256": string|null (sha256 of the pinned source file), "start_char": int >=0, "end_char": int > start_char},
 "labels": {"source_method": array of enum M_QURAN M_SUNNAH M_SAHABA M_TABIIN M_LUGHA M_QIRAAT M_NUZUL M_SIRA M_ISRAILIYYAT M_RAY (minItems 0 — a pure takhrij/content passage may have no method; but source_method and content_type together must not both be empty: express with anyOf minItems 1),"content_type": array of enum C_FIQH C_BALAGHA C_SHIR C_FADAIL C_TAKHRIJ C_NUZUL C_ISRAILIYYAT C_TAFSIR},
 "text": string — MUST equal source[start_char:end_char] exactly (Python slicing, Unicode code points),
 "verification": {"text_fidelity": enum exact|partial|modified|not_found, "strict_match": bool, "normalized_match": bool},
 "ai_proposal": {"start_char": int, "end_char": int, "labels": same shape as labels, "certainty": enum explicit|strong|weak|insufficient, "score": int 0..100, "route": enum auto_candidate|specialist},
 "review": {"status": enum ai_proposed|under_review|approved|rejected, "reviewer_role": string, "reviewed_at": date-time, "changes": array of {"at": date-time, "action": string, "from": any, "to": any}}
}
additionalProperties: false at every object level.
