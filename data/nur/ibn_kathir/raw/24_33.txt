وَلْيَسْتَعْفِفِ الَّذِينَ لَا يَجِدُونَ نِكَاحًا حَتَّى يُغْنِيَهُمُ اللَّهُ مِنْ فَضْلِهِ وَالَّذِينَ يَبْتَغُونَ الْكِتَابَ مِمَّا مَلَكَتْ أَيْمَانُكُمْ فَكَاتِبُوهُمْ إِنْ عَلِمْتُمْ فِيهِمْ خَيْرًا وَآتُوهُمْ مِنْ مَالِ اللَّهِ الَّذِي آتَاكُمْ وَلَا تُكْرِهُوا فَتَيَاتِكُمْ عَلَى الْبِغَاءِ إِنْ أَرَدْنَ تَحَصُّنًا لِتَبْتَغُوا عَرَضَ الْحَيَاةِ الدُّنْيَا وَمَنْ يُكْرِهْهُنَّ فَإِنَّ اللَّهَ مِنْ بَعْدِ إِكْرَاهِهِنَّ غَفُورٌ رَحِيمٌ (33)
وقوله تعالى {وَلْيَسْتَعْفِفِ الَّذِينَ لَا يَجِدُونَ نِكَاحًا حَتَّى يُغْنِيَهُمُ اللَّهُ مِنْ فَضْلِهِ} هذا أمر من الله تعالى لمن لا يجد تزويجًا بالتعفف عن الحرام، كما قال - صلى الله عليه وسلم -: "يا معشر الشباب من استطاع منكم الباءة فليتزوج، فإنه أغض للبصر وأحصن للفرج، ومن لم يستطع فعليه بالصوم فإنه له وجاء" الحديث ¬تقدم تخريجه في تفسير الآية نفسها.¥، وهذه الآية مطلقة، والتي في سورة النساء أخص منها، وهي قوله: {وَمَنْ لَمْ يَسْتَطِعْ مِنْكُمْ طَوْلًا أَنْ يَنْكِحَ الْمُحْصَنَاتِ} إلى أن قال: {وَأَنْ تَصْبِرُوا خَيْرٌ لَكُمْ} [النساء: 25] أي: صبركم عن تزوج الإماء خير لكم؛ لأن الولد يجيء رقيقًا {وَاللَّهُ غَفُورٌ رَحِيمٌ}.
قال عكرمة في قوله: {وَلْيَسْتَعْفِفِ الَّذِينَ لَا يَجِدُونَ نِكَاحًا} قال: هو الرجل يرى المرأة فكأنه يشتهي، فإن كانت له امرأة فليذهب إليها وليقض حاجته منها، وإن لم يكن له امرأة فلينظر في ملكوت السماوات والأرض حتى يغنيه الله ¬أخرجه ابن أبي حاتم بسند ضعيف فيه حفص بن عمر العدني، وهو ضعيف.¥.
وقوله تعالى: {وَالَّذِينَ يَبْتَغُونَ الْكِتَابَ مِمَّا مَلَكَتْ أَيْمَانُكُمْ فَكَاتِبُوهُمْ إِنْ عَلِمْتُمْ فِيهِمْ خَيْرًا} هذا أمر من الله تعالى للسادة إذا طلب عبيدهم منهم الكتابة أن يكاتبوهم بشرط أن يكون للعبد حيلة وكسب يؤدي إلى سيده المال الذي شارطه على أدائه، وقد ذهب كثير من العلماء إلى أن هذا الأمر أمر إرشاد واستحباب، لا أمر تحتم وإيجاب؛ بل السيد مخيّر إذا طلب منه عبده الكتابة، إن شاء كاتبه وإن شاء لم يكاتبه، قال الثوري عن جابر، عن الشعبي: إن شاء كاتبه وإن شاء لم يكاتبه ¬أخرجه ابن أبي حاتم من طريق الثوري به، وفي سنده جابر، وهو الجعفي: ضعيف.¥.
وكذا قال ابن وهب، عن إسماعيل بن عياش، عن رجل، عن عطاء بن أبي رباح: إن يشأ يكاتبه وإن يشأ لم يكاتبه ¬أخرجه ابن أبي حاتم من طريق ابن وهب به، وسنده ضعيف لإبهام الراوي عن عطاء بن أبي رباح، وسيأتي بسند صحيح عن عطاء.¥.
وكذا قال مقاتل بن حيان ¬أخرجه ابن أبي حاتم بسند حسن من طريق بكير بن معروف عن مقاتل بن حيان.¥ والحسن البصري.
وذهب آخرون إلى أنه يجب على السيد إذا طلب، منه عبده ذلك أن يجيبه إلى ما طلب أخذًا بظاهر هذا الأمر.
وقال البخاري: وقال روح، عن ابن جريج: قلت لعطاء: أواجب علي إذا علمت له مالًا أن أكاتبه، قال: ما أراه إلا واجبًا.
وقال عمرو بن دينار: قلت لعطاء: أتأثره عن أحد؟ قال: لا، ثم أخبرني أن موسى بن أنس أخبره أن سيرين سأل أنسًا المكاتبة، وكان كثير المال فأبى، فانطلق إلى عمر - رضي الله عنه -، فقال: كاتبه، فأبى فضربه بالدرة، ويتلو عمر - رضي الله عنه - {فَكَاتِبُوهُمْ إِنْ عَلِمْتُمْ فِيهِمْ خَيْرًا} فكاتبه. هكذا ذكره البخاري تعليقًا ¬صحيح البخاري، العتق، باب في المكاتب ونجومه في كل سنة نجم، قبل حديث رقم 2560، ووصله إسماعيل بن إسحاق القاضي في "أحكام القرآن" عن علي بن المديني عن روح به (ينظر: فتح الباري 5/ 186)، وسنده صحيح.¥، ورواه عبد الرزاق: أخبرنا ابن جريج قال: قلت لعطاء: أواجب عليّ إذا علمت له مالًا أن أكاتبه؟ قال: ما أراه إلا واجبًا وقال عمرو بن دينار، قال: قلت لعطاء: أتأثره من أحد؟ قال: لا ¬أخرجه عبد الرزاق بسنده ومتنه (المصنف 8/ 371)، وسنده صحيح، وأخرجه الطبري من طريق عبد الرزاق به.¥.
وقال ابن جرير: حدثنا محمد بن بشار، حدثنا محمد بن بكر، حدثنا سعيد، عن قتادة، عن أنس بن مالك: أن سيرين أراد أن يكاتبه، فتلكأ عليه فقال له عمر: لتكاتبنه ¬أخرجه الطبري بسنده ومتنه، وصحح سنده الحافظ ابن كثير، وهو كما قال.¥. إسناد صحيح.
وقال سعيد بن منصور: حدثنا هشيم بن جويبر، عن الضحاك قال: هي عزمة ¬سنده ضعيف لضعف جويبر.¥، وهذا القول القديم من قولي الشافعي، وذهب في الجديد إلى أنه لا يجب لقوله - عليه السلام -: "لا يحل مال امرئ مسلم إلا بطيب من نفسه" ¬أخرجه الإمام أحمد من حديث عمِّ أبي حرَّة الرقاشي مطولًا (المسند 34/ 299 - 301 ح 20695)، وقال محققوه: صحيح لغيره مقطعًا.¥.
وقال ابن وهب: قال مالك: الأمر عندنا أنه ليس على سيد العبد أن يكاتبه إذا سأله ذلك، ولم أسمع أحدًا من الأئمة أكره أحدًا على أن يكاتب عبده. قال مالك: وإنما ذلك أمر من الله تعالى وإذن منه للناس وليس بواجب ¬الموطأ 2/ 788، وأخرجه الطبري بسند صحيح من طريق ابن وهب عن الإمام مالك بنحوه.¥. وكذا قال الثوري وأبو حنيفة وعبد الرحمن بن زيد بن أسلم وغيرهم ¬أخرجه الطبري بسند صحيح من طريق ابن وهب، وهو عبد الله، عن عبد الرحمن بن زيد بن أسلم.¥، واختار ابن جرير قول الوجوب لظاهر الآية.
وقوله تعالى: {إِنْ عَلِمْتُمْ فِيهِمْ خَيْرًا} قال بعضهم: أمانة ¬أخرجه آدم بن أبي إياس بسند حسن من طريق مبارك عن الحسن.¥.
وقال بعضهم: صدقًا ¬أخرجه الطبري بسند صحيح عن الحسن، وإبراهيم النخعي.¥.
وقال بعضهم: مالًا ¬أخرجه آدم بن أبي إياس والطبري بسند صحيح من طريق ابن أبي نجيح عن مجاهد.¥. وقال بعضهم: حيلة وكسبًا ¬أخرجه ابن أبي حاتم بسند صحيح من طريق معمر عن قتادة بنحوه.¥.
وروى أبو داود في كتاب المراسيل، عن يحيى بن أبي كثير قال: قال رسول الله - صلى الله عليه وسلم -: "فكاتبوهم إن علمتم فيهم خيرًا" قال: "إن علمتم فيهم حرفة، ولا ترسلوهم كلا على الناس" ¬سنده معضل؛ لأن يحيى ابن أبي كثير تابع تابعي.¥.
وقوله تعالى: {وَآتُوهُمْ مِنْ مَالِ اللَّهِ الَّذِي آتَاكُمْ} اختلف المفسرون فيه:
فقال بعضهم: معناه اطرحوا لهم من الكتابة بعضها، ثم قال بعضهم: مقدار الربع ¬أخرجه البستي بسند حسن من طريق أبي عبد الرحمن السلمي عن علي - رضي الله عنه -.¥، وقيل: الثلث، وقيل: النصف ¬أخرجه البستي بسند ضعيف من طريق القاسم بن عبد الرحمن عن عمر - رضي الله عنه -، والقاسم لم يسمع من عمر.¥، وقيل: جزء من الكتابة من غير حد.
وقال آخرون: بل المراد من قوله: {وَآتُوهُمْ مِنْ مَالِ اللَّهِ الَّذِي آتَاكُمْ} هو النصيب الذي فرض الله لهم من أموال الزكاة، وهذا قول الحسن وعبد الرحمن بن زيد بن أسلم وأبيه ومقاتل بن حيان ¬قول الحسن أخرجه الطبري بسند صحيح من طريق يونس عنه، وقول عبد الرحمن بن زيد أخرجه ابن أبي حاتم بسند صحيح من طريق ابن وهب عنه، وقول مقاتل بن سليمان أخرجه ابن أبي حاتم بسند حسن من طريق بكير بن معروف عنه.¥، واختاره ابن جرير.
وقال إبراهيم النخعي في قوله: {وَآتُوهُمْ مِنْ مَالِ اللَّهِ الَّذِي آتَاكُمْ} قال: حثَّ الناس عليه مولاه وغيره ¬أخرجه عبد الرزاق بسند ضعيف من طريق مغيرة عن إبراهيم (المصنف 8/ 376)، ورواية مغيرة عن إبراهيم فيها مقال.¥، وكذا قال بُريدة بن الحصيب الأسلمي ¬أخرجه ابن أبي حاتم بسند جيد من طريق عبد الله بن بريدة عن أبيه.¥ وقتادة، وقال ابن عباس: أمر الله المؤمنين أن يعينوا في الرقاب. وقد تقدم في الحديث عن النبي - صلى الله عليه وسلم - أنه قال: "ثلاثة حق على الله عونهم. . ." فذكر منهم المكاتب يريد الأداء، والقول الأول أشهر.
وقال ابن أبي حاتم: حدثنا محمد بن إسماعيل، حدثنا وكيع، عن ابن شبيب، عن عكرمة، عن ابن عباس، عن عمر: أنه كاتب عبدًا له يكنى أبا أُمية، فجاء بنجمه حين حلّ فقال: يا أبا أُمية اذهب فاستعن به في مكاتبتك، فقال: يا أمير المؤمنين، لو تركته حتى يكون من آخر نجم؟ قال: أخاف أن لا أدرك ذلك، ثم قرأ {فَكَاتِبُوهُمْ إِنْ عَلِمْتُمْ فِيهِمْ خَيْرًا وَآتُوهُمْ مِنْ مَالِ اللَّهِ الَّذِي آتَاكُمْ} قال عكرمة: فكان أول نجم ¬هو القدر المعين الذي يؤديه المكاتب في وقت معين (النهاية 5/ 22).¥ أُدّي في الإسلام ¬أخرجه ابن أبي حاتم بسنده ومتنه، وسنده حسن.¥.
وقال ابن جرير: حدثنا ابن حميد، حدثنا هارون بن المغيرة، عن عنبسة، عن سالم الأفطس، عن سعيد بن جبير قال: كان ابن عمر إذا كاتب مكاتبه لم يضع عنه شيئًا من أول نجومه مخافة أن يعجز فترجع إليه صدقته، ولكنه إذا كان في آخر مكاتبته وضع عنه ما أحبّ ¬أخرجه الطبري بسنده ومتنه، وسنده ضعيف لضعف ابن حميد، وهو محمد بن حميد الرازي.¥.
وقال علي بن أبي طلحة، عن ابن عباس في الآية {وَآتُوهُمْ مِنْ مَالِ اللَّهِ الَّذِي آتَاكُمْ} قال: يعني ضعوا عنهم في مكاتبتهم ¬أخرجه الطبري بسند ثابت من طريق علي به.¥. وكذلك قال مجاهد وعطاء والقاسم بن أبي بزة وعبد الكريم بن مالك الجزري والسدي ¬أخرج بعضها الطبري وابن أبي حاتم، وتتقوى بسابقها.¥.
وقال محمد بن سيرين في قوله: {وَآتُوهُمْ مِنْ مَالِ اللَّهِ الَّذِي آتَاكُمْ} كان يعجبهم أن يدع الرجل لمكاتبه طائفة من مكاتبته ¬أخرجه ابن أبي شيبة بسند صحيح من طريق عطاء بن أبي رباح عن ابن سيرين (المصنف 6/ 371).¥، وقال ابن أبي حاتم: أخبرنا الفضل بن شاذان المقري، أخبرنا إبراهيم بن موسى، أخبرنا هشام بن يوسف، عن ابن جريج، أخبرني عطاء بن السائب: أن عبد الله بن جندُب أخبره عن علي - رضي الله عنه -، عن النبي - صلى الله عليه وسلم - قال: "ربع الكتابة" ¬أخرجه ابن أبي حاتم بسنده ومتنه، وسنده ضعيف فيه جندب: مستور، وعطاء بن السائب: صدوق اختلط، والصواب وقفه كما تقدم.¥. وهذا حديث غريب ورفعه منكر، والأشبه أنه موقوف على علي - رضي الله عنه -، كما رواه عنه أبو عبد الرحمن السلمي - رحمه الله - ¬تقدم تخريجه وثبوته عن علي - رضي الله عنه - موقوفًا في تفسير الآية نفسها.¥.
وقوله تعالى: {وَلَا تُكْرِهُوا فَتَيَاتِكُمْ عَلَى الْبِغَاءِ} الآية، كان أهل الجاهلية إذا كان لأحدهم أمة أرسلها تزني، وجعل عليها ضريبة يأخذها منها كل وقت، فلما جاء الإسلام نهى الله المؤمنين عن ذلك، وكان سبب نزول هذه الآية الكريمة، فيما ذكر غير واحد من المفسرين من السلف والخلف في شأن عبد الله بن أُبيّ بن سلول، فإنه كان له إماء، فكان يُكرههنّ على البغاء طلبًا لخراجهنَّ، ورغبة في أولادهنَّ، ورياسة منه فيما يزعم.
ذكر الآثار الواردة في ذلك:
قال الحافظ أبو بكر أحمد بن عمرو بن عبد الخالق البزار - رحمه الله - في مسنده: حدثنا أحمد بن داود الواسطي، حدثنا أبو عمرو اللخمي - يعني: محمد بن الحجاج -، حدثنا محمد بن إسحاق، وقال السدي: أُنزلت هذه الآية الكريمة في عبد الله بن أُبيّ بن سلول رأس المنافقين وكانت له جارية تدعى: مُعاذة وكان إذا نزل به ضيف أرسلها إليه ليواقعها إرادة الثواب منه والكرامة له. فأقبلت الجارية إلى أبي بكر - رضي الله عنه - فشكت إليه، فذكره أبو بكر للنبي - صلى الله عليه وسلم - فأمره بقبضها فصاح عبد الله بن أُبيّ من يعذرنا من محمد يغلبنا على مملوكتنا، فأنزل الله فيهم هذا ¬أخرجه ابن أبي حاتم بسند مرسل عن السدي ويتقوى كسابقه.¥، وقال مقاتل بن حيان: بلغني - والله أعلم - أن هذه الآية نزلت في رجلين كانا يكرهان أمتين لهما إحداهما اسمها مسيكة وكانت للأنصار، وكانت أُميمة أُم مسيكة لعبد الله بن أُبيّ وكانت معاذة وأروى بتلك المنزلة، فأتت مسيكة وأُمها النبي - صلى الله عليه وسلم - فذكرتا ذلك له، فأنزل الله في ذلك {وَلَا تُكْرِهُوا فَتَيَاتِكُمْ عَلَى الْبِغَاءِ} يعني: الزنا ¬أخرجه ابن أبي حاتم بسند حسن من طريق بكير بن معروف عن مقاتل لكنه معضل، ولبعضه شواهد يتقوى بها.¥.
وقوله تعالى: {إِنْ أَرَدْنَ تَحَصُّنًا} هذا خرج مخرج الغالب فلا مفهوم له، وقوله تعالى: {لِتَبْتَغُوا عَرَضَ الْحَيَاةِ الدُّنْيَا} أي: من خراجهن ومهورهن وأولادهن، وقد نهى رسول الله - صلى الله عليه وسلم - عن كسب الحجام ومهر البغي وحلوان الكاهن ¬أخرجه الشيخان من حديث عقبة بن عمرو - رضي الله عنه - (صحيح البخاري، البيوع، باب ثمن الكلب ح 2237)، وصحيح مسلم، المساقاة، باب تحريم ثمن الكلب (ح 1567).¥، وفي رواية: "مهر البغي خبيث، وكسب الحجام خبيث، وثمن الكلب خبيث" ¬أخرجه أبو داود من حديث رافع بن خديج - رضي الله عنه - (السنن، البيوع، باب في كسب الحجام ح 3421)، وصححه الألباني في صحيح سنن أبي داود (ح 2919).¥.
وقوله تعالى: {وَمَنْ يُكْرِهْهُنَّ فَإِنَّ اللَّهَ مِنْ بَعْدِ إِكْرَاهِهِنَّ غَفُورٌ رَحِيمٌ} أي: لهن كما تقدم في الحديث عن جابر.
وقال ابن أبي طلحة، عن ابن عباس: فإن فعلتم فإن الله لهن غفور رحيم، وإثمهن على من أكرههن ¬أخرجه الطبري وابن أبي حاتم بسند ثابت من طريق ابن أبي طلحة به.¥، وكذا قال مجاهد وعطاء الخراساني والأعمش وقتادة ¬قول مجاهد أخرجه آدم بن أبي إياس والطبري بسند صحيح من طريق ابن أبي نجيح عنه، وقول عطاء الخراساني أخرجه ابن أبي حاتم بسند ضعيف من طريق عثمان بن عطاء عن أبيه، وعثمان: ضعيف، وقول الأعمش أخرجه ابن أبي حاتم بسند صحيح من طريق ابن نمير عن الأعمش، وقول قتادة أخرجه ابن أبي حاتم بسند صحيح من طريق ابن أبي عروبة.¥.
وقال أبو عبيد: حدثني إسحاق الأزرق، عن عوف، عن الحسن في هذه الآية {فَإِنَّ اللَّهَ مِنْ بَعْدِ إِكْرَاهِهِنَّ غَفُورٌ رَحِيمٌ} قال لهنَّ: والله لهنَّ والله ¬سنده صحيح.¥.
وعن الزهري قال: غفور لهن ما أكرهن عليه.
وعن زيد بن أسلم قال: {غَفُورٌ رَحِيمٌ} للمكرهات ¬أخرجه الطبري بسند صحيح من طريق ابن وهب عن عبد الرحمن بن زيد بن أسلم.¥، حكاهن ابن المنذر في تفسيره بأسانيده.
وقال ابن أبي حاتم: حدثنا أبو زرعة، حدثنا يحيى بن عبد الله، حدثني ابن لهيعة، حدثني عطاء، عن سعيد بن جبير قال: في قراءة عبد الله بن مسعود "فإن الله من بعد إكراههن لهن غفور رحيم" وإثمهن على من أكرههن ¬أخرجه ابن أبي حاتم بسنده ومتنه، وسنده ضعيف؛ لأن سعيد بن جبير لم يسمع من ابن مسعود، والقراءة شاذة تفسيرية.¥، وفي الحديث المرفوع عن رسول الله - صلى الله عليه وسلم - أنه قال: "رفع عن أمتي الخطأ والنسيان وما استكرهوا عليه" ¬تقدم تخريجه في تفسير سورة الأعراف آية 157.¥.
