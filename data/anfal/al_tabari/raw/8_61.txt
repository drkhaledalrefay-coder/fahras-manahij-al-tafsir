<br>القولُ في تأويلِ قوله: {وَإِنْ جَنَحُوا لِلسَّلْمِ فَاجْنَحْ لَهَا وَتَوَكَّلْ عَلَى اللَّهِ إِنَّهُ هُوَ السَّمِيعُ الْعَلِيمُ (61)}.<br>يقولُ تعالى ذكرُه لنبيِّه محمدٍ - صلى الله عليه وسلم -: وإما تخافَنَّ مِن قومٍ خِيانَةً وغدرًا، فانْبِذْ إليهم على سواءٍ، وآذِنْهم بالحربِ، {وَإِنْ جَنَحُوا لِلسَّلْمِ فَاجْنَحْ لَهَا}: وإن مالوا إلى مسالمَتِك، ومُتارَكتِك الحربَ، إما بالدخولِ في الإسلامِ، وإما بإعطاءِ الجزيةِ، وإما بموادَعَةٍ، ونحوِ ذلك من أسبابِ السَّلْمِ والصلحِ، {فَاجْنَحْ لَهَا}، يقولُ: فمِلْ إليها، وابْذُل لهم ما مالوا إليه مِن ذلك وسألوكه.<br>يقالُ منه منه: جنَح الرجلُ إلى كذا يَجْنَحُ إليه جُنوحًا، وهى لتَميمٍ، وقيسٍ فيما ذُكِر عنها تقولُ: يَجْنُحُ، بضمِّ النونِ. وآخرون: يقولون: يَجْنِحُ بكسرِ النونِ، وذلك إذا مال. ومنه قولُ نابغةِ بني ذُبْيانَ ¬ديوانه ص 57.¥:<br>جَوانِحَ قَدْ أَيْقَنَّ أَن قَبِيلَه ... إذا ما الْتَقَى الجَمْعَانِ أولُ غالِبِ<br>جوانحُ: مَوائلُ.<br>وبنحوِ الذي قلْنا في ذلك قال أهلُ التأويلِ.<br>ذكرُ مَن قال ذلك<br>حدَّثنا محمدُ بنُ عبدِ الأعلى، قال: ثنا محمدُ بنُ ثورٍ، عن معمرٍ، عن قتادةَ: {وَإِنْ جَنَحُوا لِلسَّلْمِ}. قال: للصلحِ، ونسَخها قولُه: {فَاقْتُلُوا الْمُشْرِكِينَ حَيْثُ وَجَدْتُمُوهُمْ} ¬أخرجه عبد الرزاق في تفسيره 1/ 261، ومن طريقه النحاس في ناسخه ص 468 عن معمر به، وعزاه السيوطي في الدر المنثور 3/ 199 إلى ابن المنذر وأبى الشيخ.¥ [التوبة: 5].<br>حدَّثنا بشرٌ، قال: ثنا يزيدُ، قال: ثنا سعيدٌ، عن قتادةَ قولَه: {وَإِنْ جَنَحُوا لِلسَّلْمِ}: إلى الصلحِ، {فَاجْنَحْ لَهَا}. قال: وكانت هذه قبلَ "براءة"؛ كان نبيُّ اللَّهِ - صلى الله عليه وسلم - يُوادِعُ القومَ إلى أجلٍ، فإما أن يُسْلِموا، وإما أن يُقاتِلَهم ¬في م: "يقاتلوا".¥، ثم نُسِخ ذلك بعدُ في "براءة"، فقال: {فَاقْتُلُوا الْمُشْرِكِينَ حَيْثُ وَجَدْتُمُوهُمْ}. وقال: {وَقَاتِلُوا الْمُشْرِكِينَ كَافَّةً} [التوبة: 36]. ونبَذَ ¬في ص، ت 2، س، ف: "نبذوا".¥ إلى كلِّ ذى عهدٍ عهدَه ¬بعده في ص، ت 1، ت 2، س، ف: "في براءة".¥، وأمَره ¬في ص، ت 1، ت 2، س، ف: "أمرهم".¥ بقتالِهم، حتى يقولوا: لا إله إلا اللَّهُ، ويُسْلِموا، وأن لا يَقْبَل منهم إلا ذلك، وكلُّ عهدٍ كان في هذه السورةِ وفي غيرِها، وكلُّ صلحٍ يُصالِحُ به<br>المسلمون المشركين يَتَوادَعون به ¬سقط من: ص، ت 1، ت 2، س، ف.¥، فإن "براءةَ" جاءَت بنسخِ ذلك، فأُمِر بقتالِهم على كلِّ حالٍ حتى يقولوا: لا إلهَ إلا اللَّهُ.<br>حدَّثنا ابن حُميدٍ، قال: ثنا يحيى بنُ واضحٍ، عن الحسينِ ¬في النسخ: "الحسن". وتقدم هذا الإسناد كثيرا.¥، عن يزيدَ، عن عكرمةَ والحسنِ البصريِّ، قالا: {وَإِنْ جَنَحُوا لِلسَّلْمِ فَاجْنَحْ لَهَا}: نسَخَتها الآيةُ التي في "براءةَ"؛ قولُه: {قَاتِلُوا الَّذِينَ لَا يُؤْمِنُونَ بِاللَّهِ وَلَا بِالْيَوْمِ الْآخِرِ}، إلى قولِه: {وَهُمْ صَاغِرُونَ} ¬أخرجه ابن الجوزي في نواسخ القرآن ص 346 من طريق الحسين بن واقد عن يزيد عن عكرمة عن ابن عباس.¥ [التوبة: 29].<br>حدَّثني محمدُ بنُ الحسينِ، قال: ثنا أحمدُ بنُ المفضلِ، قال: ثنا أسْباطُ، عن السديِّ: {وَإِنْ جَنَحُوا لِلسَّلْمِ فَاجْنَحْ لَهَا}. يقولُ: وإن أرادوا الصلحَ فأرِدْه ¬أخرجه ابن أبي حاتم في تفسيره 5/ 1725 من طريق أحمد بن مفضل به.¥.<br>حدَّثنا ابن حميدٍ، قال: ثنا سلمةُ، عن ابن إسحاقَ: {وَإِنْ جَنَحُوا لِلسَّلْمِ فَاجْنَحْ لَهَا}، أي: إن دعَوْك إلى السَّلْمِ، إلى الإسلامِ، فصالِحْهم عليه ¬سيرة ابن هشام 1/ 674.¥.<br>حدَّثني يونُسُ، قال: أخْبرَنا ابن وهبٍ، قال: قال ابن زيدٍ في قولِه: {وَإِنْ جَنَحُوا لِلسَّلْمِ فَاجْنَحْ لَهَا}. قال: فصالِحْهم، قال: وهذا قد نسَخَه الجهادُ ¬ذكره ابن كثير في تفسيره 4/ 27.¥.<br>فأما ما قاله قتادةُ ومَن قال مثلَ قولِه مِن أن هذه الآيةَ منسوخةٌ، فقولٌ لا دَلالةَ عليه مِن كتابٍ ولا سنةٍ ولا فِطْرةِ عقلٍ.<br>وقد دلَّلْنا في غيرِ موضعٍ مِن كتابِنا هذا وغيرِه، على أن الناسخَ لا يكونُ إلا ما نفَى حكمَ المنسوخِ مِن كلِّ وجهٍ، فأما ما كان بخلافِ ذلك فغيرُ كائنٍ ناسخًا.<br>وقولُ اللَّهِ في "براءةَ": {فَاقْتُلُوا الْمُشْرِكِينَ حَيْثُ وَجَدْتُمُوهُمْ}. غيرُ نافٍ حكمُه حكمَ قولِه: {وَإِنْ جَنَحُوا لِلسَّلْمِ فَاجْنَحْ لَهَا}؛ لأن قولَه: {وَإِنْ جَنَحُوا لِلسَّلْمِ}. إنما عُنِى به بنو قريظةَ، وكانوا يهودًا أهلَ كتابٍ، وقد أذِن اللَّهُ جَلَّ ثناؤُه للمؤمنين بصلحِ أهلِ الكتابِ، ومُتارَكتِهم الحربَ، على أخْذِ الجزيةِ منهم.<br>وأما قولُه: {فَاقْتُلُوا الْمُشْرِكِينَ حَيْثُ وَجَدْتُمُوهُمْ}. فإنما عُنِى به مشركو العربِ مِن عَبَدة الأوثانِ الذين لا يَجوزُ قبولُ الجزيةِ منهم، فليس في إحدى الآيتين نفيُ حكمِ الأخرى، بل كلُّ واحدةٍ منهما مُحْكَمةٌ فيما أُنْزِلَت فيه.<br>حدَّثني محمدُ بنُ عمرٍو، قال: ثنا أبو عاصمٍ، قال: ثنا عيسى، عن ابن أبي نَجيحٍ، عن مجاهدٍ: {وَإِنْ جَنَحُوا لِلسَّلْمِ}. قال: قريظةُ ¬تفسير مجاهد ص 357، ومن طريقه ابن أبي حاتم في تفسيره 5/ 1725، وعزاه السيوطي في الدر المنثور 3/ 198 إلى ابن المنذر.¥.<br>وأما قولُه: {وَتَوَكَّلْ عَلَى اللَّهِ}. يقولُ: فوِّضْ إلى اللَّهِ يا محمدُ أمرَك، واسْتَكْفِه واثقًا به أنه يَكْفِيك.<br>كالذي حدَّثنا ابن حميدٍ، قال: ثنا سلمةُ، عن ابن إسحاقَ: {وَتَوَكَّلْ عَلَى اللَّهِ}: إن الله كافيك ¬سيرة ابن هشام 1/ 674.¥.<br>وقولُه: {إِنَّهُ هُوَ السَّمِيعُ الْعَلِيمُ}. يعنى بذلك: إن اللَّهَ الذي تَتَوَكَّلُ عليه سميعٌ لما تَقولُ أنت ومَن تُسالِمُه وتُتارِكُه الحربَ مِن أعداءِ اللَّهِ وأعدائِك، عندَ عقدِ السَّلْمِ بينَك وبينَه، وبشرطِ ¬في ص، ت 2: "يشترط"، وفى م، ت 1، س: "يشرط".¥ كلِّ فريقٍ منكم على صاحبِه مِن الشروطِ، و {الْعَلِيمُ} بما يُضْمِرُه كلُّ فريقٍ منكم للفريقِ الآخرِ مِن الوفاءِ بما عاقَدَه عليه، ومَن المُضْمِرُ ذلك منكم في قلبه، والمُنْطَوِى على خلافِه لصاحبِه ¬بعده في ت 1، س، ف: "لا رب غيره ولا معبود سواه".¥.<br>
