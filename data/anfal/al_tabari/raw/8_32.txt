<br>القولُ في تأويلِ قولِه: {وَإِذْ قَالُوا اللَّهُمَّ إِنْ كَانَ هَذَا هُوَ الْحَقَّ مِنْ عِنْدِكَ فَأَمْطِرْ عَلَيْنَا حِجَارَةً مِنَ السَّمَاءِ أَوِ ائْتِنَا بِعَذَابٍ أَلِيمٍ (32)}.<br>يقولُ تعالى ذكرُه: واذكُرْ يا محمدُ أيضًا ما حلَّ بمن قال: {اللَّهُمَّ إِنْ كَانَ هَذَا هُوَ الْحَقَّ مِنْ عِنْدِكَ فَأَمْطِرْ عَلَيْنَا حِجَارَةً مِنَ السَّمَاءِ أَوِ ائْتِنَا بِعَذَابٍ أَلِيمٍ}. إذ مكَرت لهم ¬ينظر ما تقدم فى ص 140.¥، فأتيتُهم بعذابٍ أليمٍ، وكان ذلك العذابُ قتلَهم بالسيفِ يومَ بدرٍ.<br>وهذه الآيةُ أيضًا ذُكِر أنها نزَلت في النضرِ بنِ الحارثِ.<br>ذكرُ من قال ذلك<br>حدَّثني يعقوبُ، قال: ثنا هُشيمٌ، قال: ثنا أبو بشرٍ، عن سعيدِ بنِ جُبيرٍ في قولِه: {وَإِذْ قَالُوا اللَّهُمَّ إِنْ كَانَ هَذَا هُوَ الْحَقَّ مِنْ عِنْدِكَ فَأَمْطِرْ عَلَيْنَا حِجَارَةً مِنَ السَّمَاءِ}. قال: نزَلت فى النضرِ بنِ الحارثِ ¬أخرجه سعيد بن منصور (990 - تفسير) عن هشيم به، وأخرجه ابن أبي حاتم فى تفسيره 5/ 1689 من طريق أبي بشر به¥.<br>حدَّثني محمدُ بنُ عمرٍو، قال: ثنا أبو عاصمٍ، قال: ثنا عيسى، عن ابنِ أبى نَجيحٍ، عن مجاهدٍ قولَه: {إِنْ كَانَ هَذَا هُوَ الْحَقَّ مِنْ عِنْدِكَ}. قال: قولُ النضرِ بنِ الحارثِ -[أو ابنِ الحارثِ] ¬فى م: "بن علقمة".¥ بنِ كَلَدةَ.<br>حدَّثني المثنى، قال: ثنا أبو حذيفةَ، قال: ثنا شبلٌ، عن ابنِ أبي نَجيحٍ، عن مجاهدٍ: {اللَّهُمَّ إِنْ كَانَ هَذَا هُوَ الْحَقَّ مِنْ عِنْدِكَ}: قولُ النضرِ بنِ الحارثِ ابنِ علقمةَ بنِ كَلَدةَ من بنى عبدِ الدارِ.<br>قال: أخبرنا إسحاقُ، قال: أخبرنا عبدُ اللهِ، عن وَرْقاءَ، عن ابنِ أبي نجيحٍ، عن مجاهدٍ في قولِه: {إِنْ كَانَ هَذَا هُوَ الْحَقَّ مِنْ عِنْدِكَ}. قال: هو النضرُ بنُ الحارثِ بنِ كَلَدةَ ¬تفسير مجاهد ص 453.¥.<br>حدَّثنا أحمدُ بنُ إسحاقَ، بعذابٍ أليمٍ. فقال اللهُ: {وَقَالُوا رَبَّنَا عَجِّلْ لَنَا قِطَّنَا قَبْلَ يَوْمِ الْحِسَابِ} [ص: 16]. وقال: {وَلَقَدْ جِئْتُمُونَا فُرَادَى كَمَا خَلَقْنَاكُمْ أَوَّلَ مَرَّةٍ} [الأنعام: 94]. وقال: {سَأَلَ سَائِلٌ بِعَذَابٍ وَاقِعٍ (1) لِلْكَافِرِينَ} [المعارج: 1، 2]. قال عطاءٌ: لقد نزَل فيه بِضْعَ عَشْرَةَ آيةً من كتابِ اللهِ ¬عزاه السيوطى فى الدر المنثور 3/ 181 إلى المصنف.¥.<br>حدَّثني محمدُ بنُ الحسينِ، قال: ثنا أحمدُ بنُ المفضَّلِ، قال: ثنا أسباطُ، عن السُّديِّ، قال: فقال -يعنى النضرَ بنَ الحارثِ: اللهمَّ إن كان ما يقولُ محمدٌ هو الحقَّ من عندِك، فأمطِرْ علينا حجارةً من السماءِ أو ائتِنا بعذابٍ أليمٍ. قال اللهُ: {سَأَلَ سَائِلٌ بِعَذَابٍ وَاقِعٍ (1) لِلْكَافِرِينَ} ¬أخرجه ابن أبي حاتم فى تفسيره 5/ 1690 من طريق أحمد بن المفضل به.¥.<br>حدَّثنا ابنُ حُميدٍ، قال: ثنا حكَّامٌ، عن عَنْبَسةَ، عن ليثٍ، عن مجاهدٍ في قولِه: {إِنْ كَانَ هَذَا هُوَ الْحَقَّ مِنْ عِنْدِكَ} الآية. قال: {سَأَلَ سَائِلٌ بِعَذَابٍ وَاقِعٍ (1) لِلْكَافِرِينَ}.<br>حدَّثنا بشرٌ، قال: ثنا يزيدُ، قال: ثنا سعيدٌ، عن قتادةَ قولَه: {وَإِذْ قَالُوا اللَّهُمَّ إِنْ كَانَ هَذَا هُوَ الْحَقَّ مِنْ عِنْدِكَ} الآية. قال: قال ذلك سَفَهةُ ¬فى ص، ت 2، ف: "سفه"، وفى ت 1، س: "سفيه".¥ هذه الأمةِ وجَهَلتُها، فعاد اللهُ بعائدتِه ورحمتِه على سفهةِ هذه الأمةِ وجهلتِها.<br>حدَّثنا ابنُ حُميدٍ، قال: ثنا سَلَمةُ، عن ابنِ إسحاقَ، قال: ثم ذكَر غِرَّةَ ¬فى م، ت 1، ت 2، س، ف: "غيرة". والغرة: الغفلة والاغترار. ينظر اللسان (غ ر ر).¥ قريشٍ واستِفتاحَهم على أنفسِهم: {وَإِذْ قَالُوا اللَّهُمَّ إِنْ كَانَ هَذَا هُوَ الْحَقَّ مِنْ عِنْدِكَ}. أى: ما جاء به محمدٌ، {فَأَمْطِرْ عَلَيْنَا حِجَارَةً مِنَ السَّمَاءِ} كما أمطرتَها على قومِ لوطٍ، {أَوِ ائْتِنَا بِعَذَابٍ أَلِيمٍ}. أى: ببعضِ ما عذَّبت به الأممَ قبلَنا ¬سيرة ابن هشام 1/ 670. وأخرجه ابن أبي حاتم فى تفسيره 5/ 1690، 1691. من طريق سلمة، عن ابن إسحاق، عن محمد بن جعفر، عن عروة من قوله.¥.<br>واخْتَلف أهلُ العربيةِ ¬بعده فى ص، ت 1، ت 2، س، ف: "فى قوله و"، ولعله سقط من هذه النسخ قوله: "هو الحق". ويكون السياق: في قوله: هو الحق. ووجه دخول: هو في الكلام.¥ فى وجهِ دخولِ {هُوَ} في الكلامِ؛ فقال بعضُ البصريين نُصِب {الْحَقَّ}؛ لأن {هُوَ}، واللهُ أعلمُ، حُوِّلت زائدةً في الكلامِ صلةَ توكيدٍ كزيادةِ "ما"، ولا تُزادُ إلا فى كلِّ فعلٍ لا يستغنى عن خبرٍ، ليست ¬فى م: "ليس".¥ {هُوَ} بصفةِ {هَذَا}؛ لأنك لو قلت: رأيتُ هذا هو. لم يكنْ كلامًا، ولا تكونُ "هذه" المضمر من صفةِ الظاهرةِ، ولكنها تكونُ من صفةِ المضمرةِ نحوَ قولِه: {وَلَكِنْ كَانُوا هُمُ الظَّالِمِينَ} [الزخرف: 76]، و {تَجِدُوهُ عِنْدَ اللَّهِ هُوَ خَيْرًا وَأَعْظَمَ أَجْرًا} [المزمل: 20]. لأنك تقولُ: وجَدته هو وإياى. فتكون "هو" صفةً، وقد تكونُ فى هذا المعنى أيضًا غير صفةٍ، ولكنها تكونُ زائدةً كما كان في الأوَّلِ، وقد تُجْرَى في جميعِ هذا مُجْرَى الاسمِ، فيُرْفَعُ ما بعدَها إن كان ما ¬سقط من: م.¥ بعدَها ظاهرًا أو مضمرًا في لغةِ بني تميمٍ، يقولون في قولِه: (إِن كان هذا هو الحقُّ من عندك) ¬وهي قراءة الأعمش وزيد بن على. مختصر الشواذ لابن خالويه ص 54، والبحر <br>المحيط 4/ 488.¥.<br>و (لَكِنْ كانُوا هُمُ الظَّالمونَ ¬في م، ت 1، ت 2، س، ف: "الظالمين". وقراءة الرفع هى قراءة عبد الله وأبي زيد النحويين. مختصر الشواذ لابن خالويه ص 136، والبحر المحيط 8/ 27.¥). و (تَجِدُوه عِندَ اللهِ هُوَ خَيْرٌ ¬في النسخ: "خيرا". والمثبت هو صواب استشهاد المصنف، وبالرفع قرأ أبو السمال وابن السميفع. مختصر الشواذ لابن خالويه ص 164، والبحر المحيط 8/ 367.¥ وَأَعْظَمُ أجْرًا). كما تقولُ: كانوا آباؤُهم الظالمون. جعَلوا هذا المضمرَ نحو "هو" و "هما" و "أنت" زائدًا فى هذا المكانِ، ولم تُجْعَلْ مواضعَ الصفةِ؛ لأنه فصلٌ أراد أن يبِّينَ به أنه [ليس ما بعدَه صفةً] ¬فى ص، ت 1، ت 2، س، ف: "ليس بصفة ما بعده".¥ لما قبلَه، ولم يُحتَجْ إلى هذا في الموضعِ الذى لا يكونُ له خبرٌ.<br>وكان بعضُ الكوفيين يقولُ: لم تدخُلْ "هُوَ" التى هى عمادٌ ¬تقدم معنى العماد في 2/ 214.¥ في الكلامِ إلا لمعنًى صحيحٍ. وقال: كأنه قال: زيدٌ قائمٌ. فقلتَ أنت: بل عمرٌو هو القائمُ. فـ "هو" لمعهودِ الاسمِ، والألفُ واللامُ لمعهودِ الفعلِ [والألفُ واللامُ] ¬سقط من: م. ومكانه بياض فى ص، ت 1، ت 2، س، ف. والمثبت كما أثبته الشيخ شاكر.¥ التي هي صلةٌ في الكلامِ مخالفةٌ لمعنى "هو"؛ لأن دخولَها وخروجَها واحدٌ في الكلامِ، وليست كذلك "هو"، وأما
